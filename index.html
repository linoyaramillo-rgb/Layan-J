-- Public interactions for the Layan.J photo gallery. -- Run this once in Supabase SQL Editor after reviewing the website update. create table if not exists public.gallery_likes ( id uuid primary key default gen_random_uuid(), photo_path text not null check (photo_path like 'gallery/%'), visitor_id text not null check (char_length(visitor_id) between 10 and 100), created_at timestamptz not null default now(), constraint gallery_likes_one_per_visitor unique (photo_path, visitor_id) ); create table if not exists public.gallery_comments ( id uuid primary key default gen_random_uuid(), photo_path text not null check (photo_path like 'gallery/%'), visitor_id text not null check (char_length(visitor_id) between 10 and 100), display_name text not null default 'Guest' check (char_length(display_name) between 1 and 40), comment text not null check (char_length(trim(comment)) between 1 and 500), created_at timestamptz not null default now() ); create index if not exists gallery_likes_photo_path_idx on public.gallery_likes(photo_path); create index if not exists gallery_comments_photo_created_idx on public.gallery_comments(photo_path, created_at desc); alter table public.gallery_likes enable row level security; alter table public.gallery_comments enable row level security; grant select, insert on public.gallery_likes to anon, authenticated; grant select, insert on public.gallery_comments to anon, authenticated; drop policy if exists "Anyone can view gallery likes" on public.gallery_likes; create policy "Anyone can view gallery likes" on public.gallery_likes for select to anon, authenticated using (true); drop policy if exists "Visitors can like gallery photos" on public.gallery_likes; create policy "Visitors can like gallery photos" on public.gallery_likes for insert to anon, authenticated with check ( photo_path like 'gallery/%' and char_length(visitor_id) between 10 and 100 ); drop policy if exists "Anyone can view gallery comments" on public.gallery_comments; create policy "Anyone can view gallery comments" on public.gallery_comments for select to anon, authenticated using (true); drop policy if exists "Visitors can comment on gallery photos" on public.gallery_comments; create policy "Visitors can comment on gallery photos" on public.gallery_comments for insert to anon, authenticated with check ( photo_path like 'gallery/%' and char_length(visitor_id) between 10 and 100 and char_length(display_name) between 1 and 40 and char_length(trim(comment)) between 1 and 500 );
